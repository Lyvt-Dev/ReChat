cmake_minimum_required(VERSION 3.21)
project(ReChat VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Network Svg)
qt_add_executable(ReChat src/main.cpp resources.qrc)
target_link_libraries(ReChat PRIVATE Qt6::Widgets Qt6::Network Qt6::Svg)
install(TARGETS ReChat RUNTIME DESTINATION bin BUNDLE DESTINATION .)
enable_testing()
add_test(NAME smoke COMMAND ReChat --smoke-test)
set_tests_properties(smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
